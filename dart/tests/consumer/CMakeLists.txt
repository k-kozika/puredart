cmake_minimum_required(VERSION 3.22)
project(hook_consumer LANGUAGES CXX)

enable_testing()
set(use_dart ON)
set(use_native OFF)
if(CONSUMER_CASE STREQUAL "native")
    set(use_dart OFF)
    set(use_native ON)
    add_subdirectory("${REPOSITORY}/native" native)
elseif(CONSUMER_CASE STREQUAL "native-first")
    set(use_native ON)
    add_subdirectory("${REPOSITORY}/native" native)
    add_subdirectory("${REPOSITORY}/dart" dart)
elseif(CONSUMER_CASE STREQUAL "dart-first")
    set(use_native ON)
    add_subdirectory("${REPOSITORY}/dart" dart)
    add_subdirectory("${REPOSITORY}/native" native)
elseif(CONSUMER_CASE STREQUAL "metadata")
    set(PUREDART_ENABLE_RUNTIME_HOOKS OFF CACHE BOOL "")
    add_subdirectory("${REPOSITORY}/dart" dart)
    if(TARGET purenative::core)
        message(FATAL_ERROR "Metadata-only Dart consumer unexpectedly builds native hooks")
    endif()
elseif(CONSUMER_CASE STREQUAL "dart")
    add_subdirectory("${REPOSITORY}/dart" dart)
else()
    message(FATAL_ERROR "Unknown consumer case: ${CONSUMER_CASE}")
endif()

if(PURENATIVE_BUILD_TESTS OR PUREDART_BUILD_TESTS OR
   TARGET purenative_tests OR TARGET puredart_tests)
    message(FATAL_ERROR "Embedded libraries must default to disabling their own tests")
endif()

add_executable(consumer main.cpp)
if(use_native)
    target_compile_definitions(consumer PRIVATE USE_NATIVE)
    target_link_libraries(consumer PRIVATE purenative::core)
endif()
if(use_dart)
    target_compile_definitions(consumer PRIVATE USE_DART)
    target_link_libraries(consumer PRIVATE puredart::core)
    # Exercise relative metadata paths and their default Dart-local directory.
    get_filename_component(expected_meta "${REPOSITORY}/dart/meta" ABSOLUTE)
    if(NOT PUREDART_META_DIR STREQUAL expected_meta)
        message(FATAL_ERROR "Unexpected default metadata directory: ${PUREDART_META_DIR}")
    endif()
    puredart_add_meta(consumer ../tests/meta_fixture.cpp)
    if(PUREDART_ENABLE_RUNTIME_HOOKS)
        target_compile_definitions(consumer PRIVATE USE_DART_RUNTIME)
    endif()
endif()
add_test(NAME consumer COMMAND consumer)
