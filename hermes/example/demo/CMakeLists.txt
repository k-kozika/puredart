cmake_minimum_required(VERSION 3.22)
project(purehermes_demo LANGUAGES CXX)
set(PUREHERMES_BUILD_TESTS OFF CACHE BOOL "")
if(ANDROID)
    set(PUREHERMES_BUILD_TOOLS OFF CACHE BOOL "")
else()
    set(PUREHERMES_BUILD_TOOLS ON CACHE BOOL "")
endif()
add_subdirectory(../.. purehermes)
if(ANDROID)
    if(NOT DEFINED PUREHERMES_DEMO_META OR NOT EXISTS "${PUREHERMES_DEMO_META}")
        message(FATAL_ERROR "PUREHERMES_DEMO_META must point to generated counter_meta.h; run build-demo.sh")
    endif()
    get_filename_component(meta_dir "${PUREHERMES_DEMO_META}" DIRECTORY)
    add_library(purehermes_demo SHARED demo.cpp "${PUREHERMES_DEMO_META}")
    target_include_directories(purehermes_demo PRIVATE "${meta_dir}")
    target_link_libraries(purehermes_demo PRIVATE purehermes::core log)
endif()
