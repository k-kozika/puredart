cmake_minimum_required(VERSION 3.22)
project(hermes_consumer LANGUAGES CXX)

set(PUREHERMES_BUILD_TESTS OFF CACHE BOOL "")
set(PUREDART_BUILD_TESTS OFF CACHE BOOL "")
set(PURENATIVE_BUILD_TESTS OFF CACHE BOOL "")
set(PUREHERMES_BUILD_TOOLS OFF CACHE BOOL "")
if(CONSUMER_CASE STREQUAL "reader")
    set(PUREHERMES_ENABLE_RUNTIME_HOOKS OFF CACHE BOOL "")
endif()
if(CONSUMER_CASE STREQUAL "native-first" OR CONSUMER_CASE STREQUAL "all")
    add_subdirectory("${REPOSITORY}/native" native)
endif()
add_subdirectory("${REPOSITORY}/hermes" hermes)
if(CONSUMER_CASE STREQUAL "hermes-first")
    add_subdirectory("${REPOSITORY}/native" native)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE purehermes::core)
if(CONSUMER_CASE STREQUAL "all")
    add_subdirectory("${REPOSITORY}/dart" dart)
    target_sources(consumer PRIVATE "${REPOSITORY}/dart/tests/meta_fixture.cpp")
    target_compile_definitions(consumer PRIVATE WITH_DART)
    target_link_libraries(consumer PRIVATE puredart::core purenative::core)
endif()
